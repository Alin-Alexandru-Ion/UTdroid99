cmake_minimum_required(VERSION 3.22.1)
project(UT99CxxProbe LANGUAGES C CXX)
if(NOT ANDROID OR NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "This experiment targets Android ARM64 only")
endif()
set(cxx_dir "${ANDROID_NDK}/toolchains/llvm/prebuilt/windows-x86_64/sysroot/usr/lib/aarch64-linux-android")
set(abi_archive "${CMAKE_CURRENT_BINARY_DIR}/libc++abi-ut99.a")
add_custom_command(OUTPUT "${abi_archive}"
  COMMAND "${CMAKE_OBJCOPY}" --redefine-sym __dynamic_cast=ut_original_dynamic_cast "${cxx_dir}/libc++abi.a" "${abi_archive}"
  DEPENDS "${cxx_dir}/libc++abi.a" VERBATIM)
add_custom_target(ut99_renamed_abi DEPENDS "${abi_archive}")
if(NOT GCC_BRIDGE OR NOT EXISTS "${GCC_BRIDGE}")
  message(FATAL_ERROR "Set GCC_BRIDGE to the experimental libgcc_s.so.1")
endif()
add_library(gcccandidate SHARED IMPORTED)
set_target_properties(gcccandidate PROPERTIES IMPORTED_LOCATION "${GCC_BRIDGE}")
add_library(cxxcandidate SHARED candidate.cpp)
add_dependencies(cxxcandidate ut99_renamed_abi)
set_target_properties(cxxcandidate PROPERTIES OUTPUT_NAME "stdc++" SUFFIX ".so.6")
target_link_options(cxxcandidate PRIVATE -nostdlib++ "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/versions.map" "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
target_link_libraries(cxxcandidate PRIVATE "-Wl,--whole-archive" "${cxx_dir}/libc++_static.a" "${abi_archive}" "-Wl,--no-whole-archive" dl m)
add_library(cxxthrower SHARED thrower.cpp)
target_link_options(cxxthrower PRIVATE -nostdlib++ "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
target_link_libraries(cxxthrower PRIVATE gcccandidate cxxcandidate)
add_library(cxxfixture SHARED fixture.cpp)
target_link_options(cxxfixture PRIVATE -nostdlib++ "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
target_link_libraries(cxxfixture PRIVATE cxxthrower gcccandidate cxxcandidate)
add_library(cxxinline_thrower SHARED inline_thrower.cpp)
target_link_options(cxxinline_thrower PRIVATE -nostdlib++ "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
target_link_libraries(cxxinline_thrower PRIVATE gcccandidate cxxcandidate)
add_library(cxxinline_fixture SHARED inline_fixture.cpp)
target_link_options(cxxinline_fixture PRIVATE -nostdlib++ "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
target_link_libraries(cxxinline_fixture PRIVATE cxxinline_thrower gcccandidate cxxcandidate)
