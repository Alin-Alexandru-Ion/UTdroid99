cmake_minimum_required(VERSION 3.22.1)
project(ut99host LANGUAGES CXX)
add_library(ut99host SHARED host.cpp)
target_link_libraries(ut99host android log dl)
target_link_options(ut99host PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
add_library(memfixture SHARED memfd_fixture.cpp)
target_link_options(memfixture PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
add_library(probeprovider SHARED probe_provider.cpp)
target_link_options(probeprovider PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/probe_versions.map" "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
add_library(loaderprobe SHARED probe_consumer.cpp)
target_compile_options(loaderprobe PRIVATE -fno-emulated-tls)
target_link_libraries(loaderprobe probeprovider)
target_link_options(loaderprobe PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
